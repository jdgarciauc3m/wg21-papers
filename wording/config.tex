\newcommand{\stdremove}[1]{%
  \sbox{\mybox}{#1}%
  \ifdim\wd\mybox>\linewidth
    \colorbox{red!60}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{#1}}%
  \else
    \colorbox{red!60}{#1}%
  \fi
}

  \fi
}
