\newcommand{\stdremove}[1]{%
  \sbox{\mybox}{#1}%
  \ifdim\wd\mybox>\linewidth
    \colorbox{red!60}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{\sout{#1}}}%

  \fi
